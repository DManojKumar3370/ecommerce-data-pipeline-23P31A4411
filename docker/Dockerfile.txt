# Dockerfile for e-commerce data pipeline

FROM python:3.9-slim

# Set working directory
WORKDIR /app

# Environment variables
ENV PYTHONUNBUFFERED=1 \
    PYTHONDONTWRITEBYTECODE=1 \
    PIP_NO_CACHE_DIR=1

# Install system dependencies (sqlite optional if you still use it)
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    sqlite3 \
 && rm -rf /var/lib/apt/lists/*

# Copy requirements first (better layer caching)
COPY requirements.txt .

# Install Python dependencies
RUN pip install --upgrade pip && \
    pip install -r requirements.txt

# Copy entire project
COPY . .

# Create data and log directories
RUN mkdir -p data/raw data/staging data/processed logs

# Optional: expose a port if you later add an API
EXPOSE 8000

# Simple health check (kept from original, OK even if you mainly use Postgres)
HEALTHCHECK --interval=30s --timeout=10s --start-period=40s --retries=3 \
    CMD python -c "import sqlite3; sqlite3.connect('data/ecommerce.db')"

# Default command: run the pipeline orchestrator used in this project
CMD ["python", "scripts/pipeline_orchestrator.py"]
